import {chromium} from './node_modules/playwright/index.mjs';
import {writeFile} from 'node:fs/promises';
const browser=await chromium.launch({headless:true,executablePath:'C:/Program Files/Google/Chrome/Application/chrome.exe'});
const results=[];
try {for(const width of [360,390,768,1440]){
 const context=await browser.newContext({viewport:{width,height:900},deviceScaleFactor:1});const page=await context.newPage();const errors=[];
 page.on('requestfailed',r=>errors.push({url:r.url(),error:r.failure()?.errorText}));
 await page.goto('http://127.0.0.1:4173/',{waitUntil:'networkidle'});await page.evaluate(()=>document.fonts.ready);
 results.push(await page.evaluate(()=>({width:innerWidth,fonts:[...document.fonts].map(f=>({family:f.family,status:f.status})),hero:[...document.querySelectorAll('.hero img')].map(img=>({src:img.currentSrc.split('/').pop(),naturalWidth:img.naturalWidth,renderedWidth:img.getBoundingClientRect().width,priority:img.fetchPriority})),brand:document.querySelector('.site-header .brand img').currentSrc.split('/').pop(),resources:performance.getEntriesByType('resource').filter(r=>r.name.endsWith('.woff2')||r.name.endsWith('.webp')).map(r=>({name:r.name.split('/').pop(),bytes:r.encodedBodySize,transfer:r.transferSize})),externalResources:performance.getEntriesByType('resource').filter(r=>new URL(r.name).origin!==location.origin).map(r=>r.name)})));
 results.at(-1).networkFailures=errors;await context.close();
}}finally{await browser.close();await writeFile('artifacts/interaction-polish/image-font-delivery.json',JSON.stringify(results,null,2));}
console.log(results.map(({width,fonts,hero,brand,externalResources,networkFailures})=>({width,fonts,hero,brand,externalResources,networkFailures})));
