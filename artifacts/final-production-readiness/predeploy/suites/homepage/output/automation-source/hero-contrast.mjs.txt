import{chromium}from'./node_modules/playwright/index.mjs';import{writeFile}from'node:fs/promises';
const b=await chromium.launch({headless:true,executablePath:'C:/Program Files/Google/Chrome/Application/chrome.exe'});const c=await b.newContext();const p=await c.newPage();await p.goto('http://127.0.0.1:4173/');const samples=await p.evaluate(()=>[...document.querySelectorAll('.studio-canvas *')].filter(e=>[...e.childNodes].some(n=>n.nodeType===Node.TEXT_NODE&&n.textContent.trim())).map(e=>{let bg=e;while(bg&&getComputedStyle(bg).backgroundColor==='rgba(0, 0, 0, 0)')bg=bg.parentElement;return{text:e.textContent.trim(),color:getComputedStyle(e).color,background:getComputedStyle(bg).backgroundColor};}));await b.close();
const lum=s=>s.match(/[\d.]+/g).slice(0,3).map(Number).map(v=>{v/=255;return v<=.04045?v/12.92:((v+.055)/1.055)**2.4;}).reduce((n,v,i)=>n+v*[.2126,.7152,.0722][i],0);
const results=samples.map(s=>({...s,ratio:(Math.max(lum(s.color),lum(s.background))+.05)/(Math.min(lum(s.color),lum(s.background))+.05)}));await writeFile('artifacts/homepage-premium/hero-computed-contrast.json',JSON.stringify(results,null,2));console.log(JSON.stringify({samples:results.length,minimum:Math.min(...results.map(r=>r.ratio)),failures:results.filter(r=>r.ratio<4.5)},null,2));
