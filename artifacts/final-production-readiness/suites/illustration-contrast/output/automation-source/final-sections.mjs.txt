import {chromium} from './node_modules/playwright/index.mjs';
import sharp from 'sharp';
import {mkdir,writeFile} from 'node:fs/promises';
const directory='artifacts/final-premium-review'; await mkdir(directory+'/sections',{recursive:true});
const routes=['/','/services','/services/ai-automation','/portfolio','/products','/about','/training','/contact'];
const browser=await chromium.launch({headless:true,executablePath:'C:/Program Files/Google/Chrome/Application/chrome.exe'});
const report=[];
for(const path of routes){
const name=path==='/'?'home':path.slice(1).replaceAll('/','--'); const shots={};
for(const width of [360,390,768,1024,1440,1920]){
const page=await browser.newPage({viewport:{width,height:900},reducedMotion:'reduce'});await page.goto('http://127.0.0.1:4173'+path);await page.evaluate(()=>document.fonts.ready);
await page.evaluate(async()=>{for(const img of document.images){img.scrollIntoView({behavior:'instant'});await img.decode().catch(()=>{});}scrollTo({top:0,behavior:'instant'});document.activeElement?.blur();});
const metrics=await page.evaluate(()=>{const visible=e=>e.getBoundingClientRect().width>0&&e.getBoundingClientRect().height>0;return{sections:[...document.querySelectorAll('main > section, .site-footer')].map(e=>{const r=e.getBoundingClientRect();return{class:e.className,heading:e.querySelector('h1,h2,h3')?.textContent,top:r.top+scrollY,height:r.height,background:getComputedStyle(e).backgroundColor,padding:getComputedStyle(e).paddingBlock};}),clippedText:[...document.querySelectorAll('main p, main h1, main h2, main h3, main label, main button, footer a')].filter(e=>visible(e)&&e.scrollWidth>e.clientWidth+2).map(e=>({text:e.textContent.slice(0,150),class:e.className,width:e.clientWidth,scrollWidth:e.scrollWidth})),smallTargets:[...document.querySelectorAll('main a, main button, main summary')].filter(visible).filter(e=>{const r=e.getBoundingClientRect();return r.height<24||r.width<24;}).map(e=>({text:e.textContent.trim().slice(0,100),width:e.getBoundingClientRect().width,height:e.getBoundingClientRect().height})),images:[...document.querySelectorAll('main img')].map(e=>({alt:e.alt,fit:getComputedStyle(e).objectFit,width:e.width,height:e.height,naturalWidth:e.naturalWidth,naturalHeight:e.naturalHeight}))};});report.push({path,width,...metrics});
if([390,1440].includes(width)){const file=`${directory}/sections/${name}-${width}.png`;await page.screenshot({path:file,fullPage:true,animations:'disabled'});shots[width]={file,sections:metrics.sections};}
await page.close();}
let sheet=0;let y=40;let composites=[];const flush=async()=>{if(!composites.length)return;await sharp({create:{width:1140,height:y+20,channels:3,background:'#d8e0eb'}}).composite(composites).png().toFile(`${directory}/sections/${name}-sheet-${++sheet}.png`);y=40;composites=[];};
for(let i=0;i<shots[390].sections.length;i++){
const row=[];for(const width of [1440,390]){const data=shots[width];const section=data.sections[i];if(!section)continue;const image=sharp(data.file);const meta=await image.metadata();const top=Math.max(0,Math.round(section.top));const height=Math.min(meta.height-top,Math.round(section.height));const buffer=await image.extract({left:0,top,width,height}).resize({width:width===1440?720:390}).png().toBuffer();const dim=await sharp(buffer).metadata();row.push({input:buffer,left:width===1440?10:740,top:0,height:dim.height});}
const rowHeight=Math.max(...row.map(r=>r.height));if(y+rowHeight>2900)await flush();for(const r of row){composites.push({input:r.input,left:r.left,top:y});}y+=rowHeight+20;
}await flush();console.log(`Reviewed section metrics and captured sheets: ${path}`);
}
await browser.close();await writeFile(directory+'/section-review.json',JSON.stringify(report,null,2));console.log(JSON.stringify({reviews:report.length,clipped:report.filter(r=>r.clippedText.length).map(r=>({path:r.path,width:r.width,items:r.clippedText})),smallTargets:report.filter(r=>r.smallTargets.length).map(r=>({path:r.path,width:r.width,items:r.smallTargets}))},null,2));
