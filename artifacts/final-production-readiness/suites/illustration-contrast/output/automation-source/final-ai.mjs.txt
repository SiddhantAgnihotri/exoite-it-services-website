// Optional Playwright regression: isolated .qa-tools deps and local built preview.
import { chromium } from '../.qa-tools/node_modules/playwright/index.mjs';
import { writeFile, mkdir } from 'node:fs/promises';
import assert from 'node:assert/strict';
import { aiServices, aiCategory } from '../src/data/aiServices.js';
const directory = 'artifacts/final-premium-review';
await mkdir(directory, { recursive: true });
const report = { checks: [], failures: [], mockedSubmissions: [] };
const check = (name, passed, details) => { report.checks.push({ name, passed, details }); if (!passed) report.failures.push({ name, details }); };
const browser = await chromium.launch({ headless: true, executablePath: 'C:/Program Files/Google/Chrome/Application/chrome.exe' });
const goto = (page, path) => page.goto('http://127.0.0.1:4173' + path, { waitUntil: 'domcontentloaded', timeout: 30000 });
try {
  for (const width of [360, 390, 768, 1024, 1440, 1920]) {
    const context = await browser.newContext({ viewport: { width, height: 900 } });
    let count = 0, lastService;
    await context.route('https://api.web3forms.com/submit', async route => {
      count++; lastService = route.request().postDataJSON().service;
      report.mockedSubmissions.push({ width, service: lastService, method: route.request().method() });
      await route.fulfill({ status: 200, contentType: 'application/json', body: JSON.stringify({ success: true }) });
    });
    await context.route('https://wa.me/**', route => route.fulfill({ status: 200, contentType: 'text/html', body: 'Mock only. No message sent.' }));
    const page = await context.newPage();
    await goto(page, '/');
    await page.locator('.ai-feature .button').click();
    await page.waitForURL('**/services/ai-automation');
    await page.getByRole('heading', { level: 1, name: aiServices[0].title, exact: true }).waitFor();
    check(`${width}: homepage AI feature opens automation service`, await page.getByRole('heading', { level: 1 }).textContent() === aiServices[0].title);
    if (width <= 950) {
      await page.getByRole('button', { name: 'Open menu', exact: true }).click();
      await page.locator('#mobile-menu').getByRole('link', { name: 'AI & Automation', exact: true }).click();
      check(`${width}: mobile AI navigation works and closes`, !(await page.locator('#mobile-menu').isVisible()));
    } else {
      await page.locator('.nav-disclosure summary').click();
      await page.getByRole('link', { name: aiServices[1].name, exact: true }).first().click();
      await page.waitForURL('**/services/ai-chatbots');
      check(`${width}: grouped AI navigation opens chatbot page`, !(await page.locator('.nav-services-panel').isVisible()));
    }
    await goto(page, '/services#ai-automation-solutions');
    const category = page.locator('#ai-automation-solutions');
    check(`${width}: category is crawlable and has four services`, await category.getByRole('heading', { name: aiCategory, exact: true }).isVisible() && await category.locator('article').count() === 4);
    for (const service of aiServices) {
      await goto(page, '/services/' + service.slug);
      const faq = page.locator('.faq-list details').first();
      await faq.locator('summary').focus(); await page.keyboard.press('Enter');
      check(`${width}: ${service.slug} FAQ keyboard disclosure`, await faq.getAttribute('open') !== null);
      await page.locator('.service-final-cta').getByRole('link', { name: service.cta, exact: true }).click();
      await page.waitForURL('**/contact?service=*');
      await page.waitForFunction(value => document.querySelector('#enquiry-service').value === value, service.enquiry);
      check(`${width}: ${service.slug} enquiry selects correct service`, await page.locator('#enquiry-service').inputValue() === service.enquiry);
      await page.locator('#enquiry-name').fill('Synthetic AI QA');
      await page.locator('#enquiry-email').fill('qa@example.com');
      await page.locator('#enquiry-phone').fill('+91 98765 43210');
      await page.locator('#enquiry-message').fill('Mock-only AI service verification. No real delivery.');
      const before = count;
      await page.getByRole('button', { name: 'Send enquiry', exact: true }).click();
      await page.getByText('Enquiry sent', { exact: true }).waitFor();
      check(`${width}: ${service.slug} mocked POST and success feedback`, count === before + 1 && lastService === service.enquiry && await page.locator('#enquiry-name').inputValue() === '');
    }
    await goto(page, '/services/ai-automation');
    await page.emulateMedia({ reducedMotion: 'reduce' });
    check(`${width}: reduced motion keeps concept static`, await page.locator('.ai-concept').evaluate(el => getComputedStyle(el).transitionDuration === '0s'));
    const geometry = await page.locator('.ai-concept').evaluate(el => ({ width: el.getBoundingClientRect().width, scroll: el.scrollWidth, client: el.clientWidth }));
    check(`${width}: concept fits its container`, geometry.scroll <= geometry.client + 1);
    await context.close();
    console.log(`${width}px AI flow checks completed`);
  }
} catch (error) { report.failures.push({ error: error.stack }); }
finally { await browser.close(); await writeFile(`${directory}/ai-flows.json`, JSON.stringify(report, null, 2)); }
console.log(JSON.stringify({ checks: report.checks.length, failures: report.failures }, null, 2));
assert.equal(report.failures.length, 0);
