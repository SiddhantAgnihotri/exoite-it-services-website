import{chromium}from'./node_modules/playwright/index.mjs';import{writeFile}from'node:fs/promises';
const browser=await chromium.launch({headless:true,executablePath:'C:/Program Files/Google/Chrome/Application/chrome.exe'});const context=await browser.newContext();const page=await context.newPage();const results=[];
const rgb=value=>value.startsWith('#')?value.slice(1).match(/../g).map(v=>parseInt(v,16)):value.match(/[\d.]+/g).slice(0,3).map(Number);const lum=c=>rgb(c).map(v=>{v/=255;return v<=.04045?v/12.92:((v+.055)/1.055)**2.4;}).reduce((n,v,i)=>n+v*[.2126,.7152,.0722][i],0);const ratio=(a,b)=>(Math.max(lum(a),lum(b))+.05)/(Math.min(lum(a),lum(b))+.05);
try{for(const path of ['/products','/insights']){await page.goto('http://127.0.0.1:4173'+path);await page.evaluate(()=>document.fonts.ready);const samples=await page.evaluate(()=>{const posts=[...document.querySelectorAll('.preview-art')].map(e=>({type:'poster',text:e.querySelector('strong').textContent,color:getComputedStyle(e).color,background:getComputedStyle(e).backgroundColor,decoration:getComputedStyle(e.querySelector('.poster-orbit')).borderColor,decorative:e.getAttribute('aria-hidden')}));const svg=[...document.querySelectorAll('.editorial-visual text')].map(e=>({type:'svg',text:e.textContent,color:getComputedStyle(e).fill,background:e.closest('g')?.querySelector('rect')?.getAttribute('fill')||'#ffffff',accessibleImage:e.closest('svg').getAttribute('aria-label')}));return[...posts,...svg];});results.push({path,samples:samples.map(s=>({...s,backgroundRatio:ratio(s.color,s.background),decorationRatio:s.decoration?ratio(s.color,s.decoration):null}))});}}
finally{await browser.close();await writeFile('artifacts/final-premium-review/illustration-contrast.json',JSON.stringify(results,null,2));}console.log(JSON.stringify(results,null,2));
