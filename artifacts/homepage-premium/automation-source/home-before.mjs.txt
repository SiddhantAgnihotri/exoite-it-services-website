import {chromium} from './node_modules/playwright/index.mjs';
import {mkdir,writeFile} from 'node:fs/promises';
const dir='artifacts/homepage-premium/before';await mkdir(dir,{recursive:true});
const browser=await chromium.launch({headless:true,executablePath:'C:/Program Files/Google/Chrome/Application/chrome.exe'});const results=[];
for(const width of [390,1440]){const page=await browser.newPage({viewport:{width,height:900},reducedMotion:'reduce'});await page.goto('http://127.0.0.1:4173/');await page.evaluate(async()=>{await document.fonts.ready;for(const img of document.images){img.scrollIntoView({behavior:'instant'});await img.decode().catch(()=>{});}scrollTo({top:0,behavior:'instant'});});await page.screenshot({path:`${dir}/home-${width}.png`,fullPage:true});results.push(await page.evaluate(()=>({width:innerWidth,height:document.body.scrollHeight,sections:[...document.querySelectorAll('main>section')].map(e=>({class:e.className,heading:e.querySelector('h1,h2')?.innerText,height:Math.round(e.getBoundingClientRect().height)}))})));await page.close();}
await browser.close();await writeFile(dir+'/layout.json',JSON.stringify(results,null,2));console.log(results);
