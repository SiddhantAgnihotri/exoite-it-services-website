import sharp from 'sharp';import{readFile}from'node:fs/promises';
const data=JSON.parse(await readFile('artifacts/homepage-premium/composition.json','utf8'));
for(const width of [768,1024]){const section=data.find(r=>r.width===width).sections.find(s=>s.class.includes('services-editorial'));await sharp(`artifacts/homepage-premium/home-${width}.png`).extract({left:0,top:Math.round(section.top),width,height:Math.round(section.height)}).png().toFile(`artifacts/homepage-premium/services-${width}.png`);}
