import{readFile,writeFile,readdir,mkdir,copyFile}from'node:fs/promises';import{createHash}from'node:crypto';const dir='artifacts/final-visual-review';const before=JSON.parse(await readFile(dir+'/source-before.json','utf8'));const results=[];for(const[file,hash]of Object.entries(before)){const after=createHash('sha256').update(await readFile(file)).digest('hex');results.push({file,changed:hash!==after,sha256:after});}const changed=results.filter(x=>x.changed);if(changed.length!==1||changed[0].file!=='src/styles/premium-refinement.css')throw Error('Unexpected app source changes: '+JSON.stringify(changed));await writeFile(dir+'/source-integrity.json',JSON.stringify({checkedFiles:results.length,changed,onlyWorkflowStylesChanged:true,results},null,2));await mkdir(dir+'/automation-source',{recursive:true});for(const file of await readdir('.qa-tools'))if(file.endsWith('.mjs'))await copyFile('.qa-tools/'+file,dir+'/automation-source/'+file+'.txt');console.log('Verified '+results.length+' source/configuration files; only premium-refinement.css changed.');
