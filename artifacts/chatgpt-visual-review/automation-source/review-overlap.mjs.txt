import{chromium}from'./node_modules/playwright/index.mjs';import{writeFile}from'node:fs/promises';
const b=await chromium.launch({headless:true,executablePath:'C:/Program Files/Google/Chrome/Application/chrome.exe'});const rows=[];
for(const width of[768,1024,1440]){const c=await b.newContext({viewport:{width,height:900},reducedMotion:'reduce'});const p=await c.newPage();await p.goto('http://127.0.0.1:4173/');await p.evaluate(()=>document.fonts.ready);const next=p.locator('.workflow-demo-actions button');await next.evaluate(e=>{const r=e.getBoundingClientRect();scrollTo({top:r.top+scrollY+r.height/2-852,behavior:'instant'});});const geometry=await p.evaluate(()=>{const a=document.querySelector('.workflow-demo-actions button'),b=document.querySelector('.whatsapp-button');const ar=a.getBoundingClientRect(),br=b.getBoundingClientRect();const left=Math.max(ar.left,br.left),right=Math.min(ar.right,br.right),top=Math.max(ar.top,br.top),bottom=Math.min(ar.bottom,br.bottom);const hit=right>left&&bottom>top?document.elementFromPoint((left+right)/2,(top+bottom)/2):null;return{scrollY,next:{x:ar.x,y:ar.y,width:ar.width,height:ar.height},whatsapp:{x:br.x,y:br.y,width:br.width,height:br.height},intersection:{width:Math.max(0,right-left),height:Math.max(0,bottom-top)},hit:hit?.closest('a,button')?.getAttribute('class')};});rows.push({width,...geometry});await p.screenshot({path:`artifacts/chatgpt-visual-review/interactions/problem-whatsapp-overlap-${width}.png`});await c.close();}
await b.close();await writeFile('artifacts/chatgpt-visual-review/overlap-observations.json',JSON.stringify(rows,null,2));console.log(rows);
