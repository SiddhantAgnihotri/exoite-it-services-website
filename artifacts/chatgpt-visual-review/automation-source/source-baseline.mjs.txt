import {readdir,readFile,writeFile,mkdir} from 'node:fs/promises';import {createHash} from 'node:crypto';
await mkdir('artifacts/chatgpt-visual-review',{recursive:true});
const files=['package.json','package-lock.json','index.html','vercel.json'];async function walk(dir){for(const e of await readdir(dir,{withFileTypes:true})){const p=dir+'/'+e.name;if(e.isDirectory())await walk(p);else files.push(p);}}for(const dir of ['src','public','scripts'])await walk(dir);
const hashes={};for(const file of files)hashes[file]=createHash('sha256').update(await readFile(file)).digest('hex');await writeFile('artifacts/chatgpt-visual-review/source-hashes-before.json',JSON.stringify(hashes,null,2));console.log('Recorded '+files.length+' source/configuration hashes.');
