import{chromium}from'./node_modules/playwright/index.mjs';import{readFile,writeFile}from'node:fs/promises';
const dir='artifacts/chatgpt-visual-review';const desktop=[];const b=await chromium.launch({headless:true,executablePath:'C:/Program Files/Google/Chrome/Application/chrome.exe'});
for(const width of[768,1024,1440]){const c=await b.newContext({viewport:{width,height:900}});const p=await c.newPage();await p.goto('http://127.0.0.1:4173/');await p.evaluate(()=>document.fonts.ready);const record=(name,passed,details)=>desktop.push({width,name,passed,details});record('Fine-pointer context supports hover',await p.evaluate(()=>matchMedia('(hover:hover) and (pointer:fine)').matches));
const s=p.locator('.home-service').first();await s.scrollIntoViewIfNeeded();await p.waitForTimeout(550);await s.hover();await p.waitForTimeout(300);const motion=await s.evaluate(e=>({transform:getComputedStyle(e).transform,duration:getComputedStyle(e).transitionDuration}));record('Service hover lifts three pixels',Math.abs(Number(motion.transform.split(',').at(-1)?.replace(')',''))+3)<.2,motion);await p.screenshot({path:`${dir}/interactions/service-hover-${width}.png`});
const w=p.locator('.work-feature').first();await w.scrollIntoViewIfNeeded();await p.waitForTimeout(550);await w.hover();await p.waitForTimeout(300);const image=await w.locator('img').evaluate(e=>{const s=getComputedStyle(e),r=e.getBoundingClientRect(),b=e.parentElement.getBoundingClientRect();return{scale:new DOMMatrix(s.transform).a,inside:r.left>=b.left&&r.right<=b.right&&r.top>=b.top&&r.bottom<=b.bottom};});record('Portfolio hover zoom is restrained and contained',Math.abs(image.scale-1.025)<.002&&image.inside,image);await p.screenshot({path:`${dir}/interactions/portfolio-hover-${width}.png`});
await p.locator('.digital-studio').hover();await p.waitForTimeout(300);record('Hero layer hover moves four pixels',await p.locator('.studio-website').evaluate(e=>Math.abs(new DOMMatrix(getComputedStyle(e).transform).m42+4)<.2));
await p.emulateMedia({reducedMotion:'reduce'});await s.hover();record('Reduced motion removes desktop hover',await s.evaluate(e=>getComputedStyle(e).transform==='none'&&getComputedStyle(e).transitionDuration==='0s'));await c.close();}
await b.close();const prior=JSON.parse(await readFile(dir+'/interaction-touch-context-raw.json','utf8'));const inapplicable=prior.results.filter(r=>r.name.includes('service hover lifts')||r.name.includes('portfolio hover zoom'));const relevant=prior.results.filter(r=>!inapplicable.includes(r));const report={results:relevant,desktop,notes:prior.notes,initialProbeCorrection:{reason:'The initial hover probes ran in hasTouch:true contexts, where hover:none is expected. Those six probes are inapplicable to desktop hover; fresh fine-pointer desktop contexts were used instead.',inapplicable}};await writeFile(dir+'/interaction-observations.json',JSON.stringify(report,null,2));console.log(JSON.stringify({relevant:relevant.length,desktop:desktop.length,failures:[...relevant,...desktop].filter(r=>!r.passed)}));
