import{chromium}from'./node_modules/playwright/index.mjs';import{writeFile,mkdir,copyFile,readdir,stat}from'node:fs/promises';
const d='artifacts/chatgpt-visual-review';const b=await chromium.launch({headless:true,executablePath:'C:/Program Files/Google/Chrome/Application/chrome.exe'});const results=[];
for(const width of[390,1440]){const c=await b.newContext({viewport:{width,height:900},reducedMotion:'reduce'});const p=await c.newPage();await p.goto('http://127.0.0.1:4173/');await p.evaluate(()=>document.fonts.ready);await p.keyboard.press('Tab');await p.keyboard.press('Enter');await p.keyboard.press('Tab');results.push({width,focused:await p.evaluate(()=>({text:document.activeElement.innerText,focusVisible:document.activeElement.matches(':focus-visible'),outline:getComputedStyle(document.activeElement).outline}))});await p.screenshot({path:`${d}/interactions/keyboard-tab-focus-${width}.png`});await c.close();}await b.close();await mkdir(d+'/asset-references',{recursive:true});for(const name of['inventory','exam-portal','lms'])await copyFile(`src/assets/images/optimized/portfolio-${name}-1280.webp`,`${d}/asset-references/portfolio-${name}-1280.webp`);const assets=[];for(const f of await readdir('src/assets/images/optimized'))if(/portfolio-(inventory|exam-portal|lms)-.*webp/.test(f))assets.push({file:f,bytes:(await stat('src/assets/images/optimized/'+f)).size});await writeFile(d+'/supplementary-observations.json',JSON.stringify({keyboard:results,assets},null,2));console.log(results);
