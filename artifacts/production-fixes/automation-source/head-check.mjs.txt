import {chromium} from './node_modules/playwright/index.mjs';
import {writeFile,readFile} from 'node:fs/promises';
import {pages,origin} from '../src/data/seo.js';
const browser=await chromium.launch({headless:true,executablePath:'C:/Program Files/Google/Chrome/Application/chrome.exe'});
const context=await browser.newContext();const page=await context.newPage();const results=[];
try{for(const [path,meta]of Object.entries(pages)){
await page.goto('http://127.0.0.1:4173'+path,{waitUntil:'domcontentloaded',timeout:30000});await page.evaluate(()=>new Promise(resolve=>requestAnimationFrame(()=>requestAnimationFrame(resolve))));
const head=await page.evaluate(()=>({titles:[...document.querySelectorAll('title')].map(e=>e.textContent),description:[...document.querySelectorAll('meta[name=description]')].map(e=>e.content),robots:[...document.querySelectorAll('meta[name=robots]')].map(e=>e.content),canonical:[...document.querySelectorAll('link[rel=canonical]')].map(e=>e.href),schemas:[...document.querySelectorAll('script[type="application/ld+json"]')].map(e=>JSON.parse(e.textContent))}));
results.push({path,passed:head.titles.length===1&&head.titles[0]===meta.title&&head.description.length===1&&head.description[0]===meta.description&&head.canonical.length===1&&head.canonical[0]===origin+path&&head.robots.length===1&&head.robots[0]===(meta.noindex?'noindex,follow':'index,follow')&&head.schemas.length===1,head});}
}finally{await browser.close();await writeFile('artifacts/production-fixes/hydrated-metadata.json',JSON.stringify(results,null,2));console.log('Hydrated metadata:',results.filter(r=>r.passed).length+'/'+results.length,'passed');console.log('Failures:',JSON.stringify(results.filter(r=>!r.passed)));}
