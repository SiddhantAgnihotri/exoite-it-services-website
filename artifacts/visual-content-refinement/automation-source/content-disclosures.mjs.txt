import{chromium}from'./node_modules/playwright/index.mjs';import{writeFile}from'node:fs/promises';import{serviceDetails}from'../src/data/serviceDetails.js';const b=await chromium.launch({headless:true,executablePath:'C:/Program Files/Google/Chrome/Application/chrome.exe'});const out=[];for(const width of[360,768,1440]){const p=await b.newPage({viewport:{width,height:900},reducedMotion:'reduce'});for(const service of serviceDetails){await p.goto('http://127.0.0.1:4173/services/'+service.slug);await p.evaluate(()=>document.fonts.ready);const heroLines=await p.locator('.page-hero-description').evaluate(e=>{const r=document.createRange();r.selectNodeContents(e);return[...r.getClientRects()].filter(x=>x.width>0).length;});const d=p.locator('.service-scope-summary .content-detail');await d.locator('summary').focus();await p.keyboard.press('Enter');const visible=await d.locator('p').first().isVisible();const focused=await d.locator('summary').evaluate(e=>e===document.activeElement&&getComputedStyle(e).outlineWidth==='3px');out.push({width,slug:service.slug,heroLines,scopeOpen:visible,keyboardFocus:focused});if(service.family==='ai'){for(const detail of await p.locator('.ai-consideration-grid .content-detail').all()){await detail.locator('summary').focus();await p.keyboard.press('Enter');out.push({width,slug:service.slug,technicalScopeVisible:await detail.locator('p').isVisible()});}}}await p.close();}await b.close();await writeFile('artifacts/visual-content-refinement/disclosure-regressions.json',JSON.stringify(out,null,2));const failures=out.filter(x=>x.heroLines>2||x.scopeOpen===false||x.keyboardFocus===false||x.technicalScopeVisible===false);console.log({checks:out.length,failures});if(failures.length)process.exitCode=1;
