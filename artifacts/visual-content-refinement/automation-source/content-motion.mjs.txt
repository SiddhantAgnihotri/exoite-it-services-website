import { mkdir, writeFile } from 'node:fs/promises';

const directory = 'artifacts/visual-content-refinement/motion';
await mkdir(directory, { recursive: true });
const report = { checks: [], failures: [], touch: [], performance: {} };
const check = (name, passed, details) => { report.checks.push({ name, passed, details }); if (!passed) report.failures.push({ name, details }); };
const browser = await chromium.launch({ headless: true, executablePath: 'C:/Program Files/Google/Chrome/Application/chrome.exe' });
const goto = (page, path = '/') => page.goto('http://127.0.0.1:4173' + path, { waitUntil: 'domcontentloaded', timeout: 30000 });
const settle = async locator => { await locator.scrollIntoViewIfNeeded(); await locator.page().waitForTimeout(520); };
const translateY = value => value === 'none' ? 0 : Number(value.match(/matrix\([^,]+,[^,]+,[^,]+,[^,]+,[^,]+,\s*([^\)]+)/)?.[1]);
try {
  const context = await browser.newContext({ viewport: { width: 1440, height: 900 } });
  await context.addInitScript(() => {
    window.qaLayoutShifts = [];
    new PerformanceObserver(list => { for (const entry of list.getEntries()) if (!entry.hadRecentInput) window.qaLayoutShifts.push(entry.value); }).observe({ type: 'layout-shift', buffered: true });
  });
  const page = await context.newPage();
  await goto(page); await page.evaluate(() => document.fonts.ready); await page.waitForTimeout(550);
  const button = page.locator('.hero .button-primary');
  await button.hover(); await page.waitForTimeout(300);
  const buttonStyle = await button.evaluate(el => ({ transform: getComputedStyle(el).transform, duration: getComputedStyle(el).transitionDuration, properties: getComputedStyle(el).transitionProperty }));
  check('Button lift uses a 180â€“300ms transform transition', Math.abs(translateY(buttonStyle.transform) + 3) < .2 && buttonStyle.properties.includes('transform') && buttonStyle.duration.includes('0.22s'), buttonStyle);
  await page.mouse.move(0, 0); await button.focus();
  check('Keyboard focus has a clear three-pixel outline', await button.evaluate(el => getComputedStyle(el).outlineStyle === 'solid' && getComputedStyle(el).outlineWidth === '3px'));
  await page.screenshot({ path: `${directory}/hero-focus-1440.png`, animations: 'disabled' });
  const heroFrame = page.locator('.digital-studio'); await heroFrame.hover(); await page.waitForTimeout(300);
  check('Hero has gentle layered hover depth', await page.locator('.studio-website').evaluate(el => Math.abs(new DOMMatrix(getComputedStyle(el).transform).m42 + 4) < .2));
  await page.screenshot({ path: `${directory}/hero-hover-1440.png`, animations: 'disabled' });
  for (const [label, selector, lift] of [['Service row', '.home-service', 0], ['Portfolio card', '.work-feature', 3], ['Product preview', '.software-preview', 4], ['Article card', '.insight-lead', 3]]) {
    const element = page.locator(selector).first(); await settle(element); await element.hover(); await page.waitForTimeout(300);
    const transform = await element.evaluate(el => getComputedStyle(el).transform);
    check(`${label} has restrained hover lift`, Math.abs(translateY(transform) + lift) < .2, transform);
    if (label === 'Portfolio card') {
      const image = element.locator('.image-stage img');
      const scale = await image.evaluate(el => new DOMMatrix(getComputedStyle(el).transform).a);
      check('Portfolio zoom remains within 1.02â€“1.04', scale >= 1.02 && scale <= 1.04, scale);
      check('Zoomed image remains inside the padded stage', await image.evaluate(el => { const r = el.getBoundingClientRect(), stage = el.parentElement.getBoundingClientRect(); return r.left >= stage.left && r.right <= stage.right && r.top >= stage.top && r.bottom <= stage.bottom; }));
    }
  }
  const cardLink = page.locator('.home-service .text-link').first(); await cardLink.focus();
  await page.keyboard.press('Tab'); await page.keyboard.press('Shift+Tab');
  check('Service link has keyboard focus on the actual target', await cardLink.evaluate(el => el === document.activeElement && getComputedStyle(el).outlineWidth === '3px'));
  const industry = page.locator('.industry-rows details').nth(1); await settle(industry); await industry.locator('summary').hover(); await page.waitForTimeout(240);
  check('Industry item has gentle background feedback', await industry.evaluate(el => getComputedStyle(el).backgroundColor === 'rgb(248, 250, 252)'));
  await industry.locator('summary').focus(); await page.keyboard.press('Enter');
  check('Industry accordion opens with Enter', await industry.getAttribute('open') !== null);
  const software = page.locator('.software-preview'); await settle(software);
  await software.getByRole('button', { name: 'Content', exact: true }).click();
  check('Product controls stay visible and functional', await software.getByRole('button', { name: 'Content', exact: true }).getAttribute('aria-pressed') === 'true');

  // A new page starts with an untouched observer and no scrolled/focused state.
  const revealPage = await context.newPage(); await goto(revealPage); await revealPage.evaluate(() => document.fonts.ready);
  const target = revealPage.locator('.services-editorial > .container');
  await target.evaluate(el => window.scrollTo({ top: el.getBoundingClientRect().top + scrollY - 280, behavior: 'instant' }));
  await revealPage.waitForFunction(() => document.querySelector('.services-editorial > .container').getAnimations().length > 0);
  const reveal = await target.evaluate(el => { const animation = el.getAnimations()[0]; return { frames: animation.effect.getKeyframes(), duration: animation.effect.getTiming().duration, opacity: getComputedStyle(el).opacity }; });
  check('Section reveal is bounded and never fades readable text', reveal.duration === 420 && reveal.opacity === '1' && reveal.frames[0].transform === 'translateY(12px)' && !('opacity' in reveal.frames[0]), reveal);
  await revealPage.locator('.home-service .text-link').first().focus();
  check('Focus cancels moving section immediately', await target.evaluate(el => el.getAnimations().length === 0));
  await revealPage.evaluate(() => scrollTo({ top: 0, behavior: 'instant' })); await revealPage.waitForTimeout(60);
  await target.evaluate(el => scrollTo({ top: el.getBoundingClientRect().top + scrollY - 280, behavior: 'instant' })); await revealPage.waitForTimeout(100);
  check('A revealed section does not repeat on return', await target.evaluate(el => el.getAnimations().length === 0));
  await goto(revealPage, '/services');
  check('Route navigation removes old reveal animations', await revealPage.evaluate(() => document.getAnimations().filter(a => a.playState === 'running').length === 0));
  await revealPage.close();

  await page.emulateMedia({ reducedMotion: 'reduce' }); await goto(page); await page.evaluate(() => document.fonts.ready);
  await page.locator('.home-service').first().hover();
  check('Reduced motion removes hover transforms and transitions', await page.locator('.home-service').first().evaluate(el => getComputedStyle(el).transform === 'none' && getComputedStyle(el).transitionDuration === '0s'));
  check('Reduced motion has no running CSS or section animation', await page.evaluate(() => document.getAnimations().filter(a => a.playState === 'running').length === 0));
  await page.screenshot({ path: `${directory}/reduced-motion-1440.png`, fullPage: true, animations: 'disabled' });
  await page.emulateMedia({ reducedMotion: 'no-preference' });
  await page.locator('.work-feature').first().scrollIntoViewIfNeeded();
  await page.emulateMedia({ reducedMotion: 'reduce' });
  check('Changing motion preference cancels active reveals', await page.evaluate(() => document.getAnimations().filter(a => a.playState === 'running').length === 0));
  report.performance.localLayoutShiftSum = await page.evaluate(() => window.qaLayoutShifts.reduce((sum, value) => sum + value, 0));
  check('Local non-input layout-shift sum is within 0.1', report.performance.localLayoutShiftSum <= .1, report.performance.localLayoutShiftSum);
  await context.close();

  for (const width of [360, 390, 768]) {
    const touchContext = await browser.newContext({ viewport: { width, height: 900 }, isMobile: true, hasTouch: true, deviceScaleFactor: 2 });
    const touchPage = await touchContext.newPage(); await goto(touchPage);
    check(`${width}: touch device does not depend on hover`, await touchPage.evaluate(() => matchMedia('(hover: none)').matches));
    await touchPage.getByRole('button', { name: 'Open menu', exact: true }).tap();
    check(`${width}: mobile menu opens on tap`, await touchPage.locator('#mobile-menu').isVisible());
    await touchPage.locator('#mobile-menu').getByRole('link', { name: 'Services', exact: true }).tap(); await touchPage.waitForURL('**/services');
    check(`${width}: mobile navigation closes after tapping a route`, !(await touchPage.locator('#mobile-menu').isVisible()));
    await goto(touchPage); await touchPage.locator('.faq-list details summary').first().tap();
    check(`${width}: FAQ works by touch`, await touchPage.locator('.faq-list details').first().getAttribute('open') !== null);
    await touchPage.locator('.home-service h3 a[href="/services/website-development"]').tap(); await touchPage.waitForURL('**/services/website-development');
    await touchPage.getByRole('link', { name: 'Enquire Now', exact: true }).first().tap(); await touchPage.waitForURL('**/contact?service=*');
    await touchPage.waitForFunction(() => document.querySelector('#enquiry-service').value === 'Website Development');
    check(`${width}: visible card/CTA path works without hover`, await touchPage.locator('#enquiry-service').inputValue() === 'Website Development');
    check(`${width}: no float obscures Contact controls`, await touchPage.locator('.whatsapp-button').count() === 0 && await touchPage.evaluate(() => document.documentElement.scrollWidth <= innerWidth));
    await touchPage.screenshot({ path: `${directory}/touch-contact-${width}.png`, fullPage: true, animations: 'disabled' });
    report.touch.push({ width, deviceScaleFactor: 2, hasTouch: true }); await touchContext.close();
  }
  const noJS = await browser.newContext({ javaScriptEnabled: false, viewport: { width: 390, height: 900 } });
  const noJSPage = await noJS.newPage(); await goto(noJSPage);
  check('No-JavaScript homepage content and CTAs remain visible', await noJSPage.getByRole('heading', { level: 1 }).isVisible() && await noJSPage.locator('.hero .button-primary').isVisible() && await noJSPage.locator('.ai-feature h2').isVisible());
  await noJSPage.screenshot({ path: `${directory}/no-javascript-home-390.png`, fullPage: true, animations: 'disabled' }); await noJS.close();
  const fallback = await browser.newContext(); await fallback.addInitScript(() => { window.IntersectionObserver = undefined; });
  const fallbackPage = await fallback.newPage(), errors = []; fallbackPage.on('pageerror', error => errors.push(error.message)); await goto(fallbackPage);
  check('Missing IntersectionObserver leaves content functional', await fallbackPage.getByRole('heading', { level: 1 }).isVisible() && errors.length === 0, errors); await fallback.close();
} catch (error) { report.failures.push({ error: error.stack }); }
finally { await browser.close(); await writeFile(`${directory}/interaction-results.json`, JSON.stringify(report, null, 2)); }
console.log(JSON.stringify({ checks: report.checks.length, failures: report.failures, performance: report.performance }, null, 2));
assert.equal(report.failures.length, 0);
