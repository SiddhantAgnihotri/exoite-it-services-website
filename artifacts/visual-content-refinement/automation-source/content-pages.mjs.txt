import{chromium}from'./node_modules/playwright/index.mjs';import sharp from 'sharp';import{writeFile,mkdir,readFile}from'node:fs/promises';import{pages}from'../src/data/seo.js';const dir='artifacts/visual-content-refinement';await mkdir(dir+'/pages',{recursive:true});await mkdir(dir+'/details',{recursive:true});const b=await chromium.launch({headless:true,executablePath:'C:/Program Files/Google/Chrome/Application/chrome.exe'});const results=[];const baseline=JSON.parse(await readFile(dir+'/baseline.json','utf8'));const slug=p=>p==='/'?'home':p.slice(1).replaceAll('/','--');for(const width of[390,1440]){const p=await b.newPage({viewport:{width,height:900},reducedMotion:'reduce'});for(const path of Object.keys(pages)){await p.goto('http://127.0.0.1:4173'+path);await p.evaluate(()=>document.fonts.ready);const m=await p.evaluate(()=>{const visible=e=>e.checkVisibility()&&!e.closest('noscript,[aria-hidden=true]');const ps=[...document.querySelectorAll('main p')].filter(visible);const hero=document.querySelector('.page-hero-description,.hero-copy > p:not(.eyebrow):not(.hero-footnote)');const range=document.createRange();if(hero)range.selectNodeContents(hero);return{height:document.documentElement.scrollHeight,proseWords:ps.reduce((s,e)=>s+e.innerText.trim().split(/\s+/).length,0),paragraphs:ps.length,paragraphsOver30Words:ps.filter(e=>e.innerText.trim().split(/\s+/).length>30).length,title:document.title,canonical:document.querySelector('link[rel=canonical]')?.href,heroLines:hero?[...range.getClientRects()].filter(r=>r.width>0).length:0,heroText:hero?.innerText};});const before=baseline.find(x=>x.width===width&&x.path===path);results.push({width,path,...m,beforeProseWords:before?.proseWords,proseReduction:before?before.proseWords-m.proseWords:null,metadataPreserved:!before||(before.title===m.title&&before.canonical===m.canonical)});if(!pages[path].noindex){await p.evaluate(async()=>{for(const img of document.images){img.scrollIntoView({behavior:'instant'});await img.decode().catch(()=>{});}scrollTo({top:0,behavior:'instant'});document.activeElement.blur();});const full=`${dir}/pages/${slug(path)}-${width}.png`;await p.screenshot({path:full,fullPage:true,animations:'disabled'});const regions=await p.evaluate(()=>[...document.querySelectorAll('main > section')].slice(0,2).map(e=>{const r=e.getBoundingClientRect();return{top:Math.floor(r.top+scrollY),height:Math.min(1200,Math.ceil(r.height))};}));for(let i=0;i<regions.length;i++)await sharp(full).extract({left:0,top:regions[i].top,width,height:regions[i].height}).png().toFile(`${dir}/details/${slug(path)}-section-${i+1}-top-${width}.png`);}}await p.close();console.log('Captured internal pages and text metrics at '+width+'px');}await b.close();await writeFile(dir+'/content-comparison.json',JSON.stringify(results,null,2));console.log('Metadata changes',results.filter(r=>!r.metadataPreserved));console.log('Hero descriptions over two lines',results.filter(r=>r.heroLines>2&&!pages[r.path].noindex).map(r=>({width:r.width,path:r.path,lines:r.heroLines,text:r.heroText})));
