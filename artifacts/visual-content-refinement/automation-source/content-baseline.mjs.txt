import{chromium}from'./node_modules/playwright/index.mjs';import{writeFile,readdir,readFile,mkdir}from'node:fs/promises';import{createHash}from'node:crypto';import{pages}from'../src/data/seo.js';
const dir='artifacts/visual-content-refinement';const files=['package.json','package-lock.json','index.html','vercel.json'];async function walk(d){for(const e of await readdir(d,{withFileTypes:true})){if(e.isDirectory())await walk(d+'/'+e.name);else files.push(d+'/'+e.name);}}for(const d of['src','public','scripts'])await walk(d);const hashes={};for(const f of files)hashes[f]=createHash('sha256').update(await readFile(f)).digest('hex');await writeFile(dir+'/source-before.json',JSON.stringify(hashes,null,2));
const b=await chromium.launch({headless:true,executablePath:'C:/Program Files/Google/Chrome/Application/chrome.exe'});const results=[];const screenshots=['/','/services','/services/ai-automation','/services/ai-chatbots','/about','/portfolio','/products','/training','/contact'];const slug=p=>p==='/'?'home':p.slice(1).replaceAll('/','--');for(const width of[390,1440]){const p=await b.newPage({viewport:{width,height:900},reducedMotion:'reduce'});for(const path of(width===1440?Object.keys(pages):screenshots)){await p.goto('http://127.0.0.1:4173'+path);await p.evaluate(()=>document.fonts.ready);const metrics=await p.evaluate(()=>{const visible=e=>e.checkVisibility()&&!e.closest('noscript,[aria-hidden=true]');const ps=[...document.querySelectorAll('main p')].filter(visible);return{height:document.documentElement.scrollHeight,proseWords:ps.reduce((s,e)=>s+e.innerText.trim().split(/\s+/).length,0),paragraphs:ps.length,paragraphsOver30Words:ps.filter(e=>e.innerText.trim().split(/\s+/).length>30).length,title:document.title,canonical:document.querySelector('link[rel=canonical]')?.href};});results.push({width,path,...metrics});if(screenshots.includes(path)){await p.evaluate(async()=>{for(const img of document.images){img.scrollIntoView({behavior:'instant'});await img.decode().catch(()=>{});}scrollTo({top:0,behavior:'instant'});});await p.screenshot({path:`${dir}/before/${slug(path)}-${width}.png`,fullPage:true,animations:'disabled'});}}await p.close();}await b.close();await writeFile(dir+'/baseline.json',JSON.stringify(results,null,2));console.log('Saved source baseline, 18 screenshots and route prose metrics.');
